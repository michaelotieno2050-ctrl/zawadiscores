<script>
const supabaseUrl = 'https://ysnzznmbbminmhxnqwls.supabase.co/rest/v1/';
const supabaseKey = 'sb_publishable_GfsXQo4xdM7SOx-0Tk0Ikg_PswFjQPN';
const supabase = window.supabase.createClient(supabaseUrl, supabaseKey);

async function signIn(){
  await supabase.auth.signInWithOAuth({provider:'google'});
}
async function showUpload(){
  const {data} = await supabase.auth.getUser();
  if(!data.user){ alert('Please Sign In to upload. Downloading is free for everyone.'); signIn(); return; }
  document.getElementById('uploadBox').style.display='block';
}
async function loadScores(){
  let q = document.getElementById('search').value;
  let {data} = await supabase.from('scores').select('*').order('created_at',{ascending:false});
  if(q) data = data.filter(s=> (s.title+s.composer).toLowerCase().includes(q.toLowerCase()));
  document.getElementById('list').innerHTML = data.map(s=>`
  <div class="card"><b>${s.title}</b> - ${s.composer}<br>
  ${s.category} • ${s.voicing}<br><br>
  <a href="${s.pdf_url}" target="_blank"><button>Free Download PDF</button></a>
  </div>`).join('') || '<p style="text-align:center">No scores yet. Be the first to upload!</p>';
}
async function uploadScore(){
  const {data:{user}} = await supabase.auth.getUser();
  const file = document.getElementById('pdfFile').files[0];
  let pdf_url = '';
  if(file){
    let name = Date.now()+'_'+file.name;
    await supabase.storage.from('sheets').upload(name, file);
    pdf_url = supabase.storage.from('sheets').getPublicUrl(name).data.publicUrl;
  }
  await supabase.from('scores').insert({
    title: document.getElementById('title').value,
    composer: document.getElementById('composer').value,
    category: document.getElementById('category').value,
    voicing: document.getElementById('voicing').value,
    pdf_url: pdf_url,
    uploader_id: user.id,
    uploader_name: user.email
  });
  alert('Published! Thank you for the gift.');
  loadScores();
}
loadScores();
</script>
